var afn_getRMSEandR2 = function(linearTrend, imageCollection, Independents, dependent) {
    // RMSE stuff pulled from here:
    // https://gis.stackexchange.com/questions/394741/calculating-r-squared-using-the-residuals-band-from-google-earth-engines-line
    imageCollection = ee.ImageCollection(imageCollection)
    // Compute the number of observations in each pixel.
    var n = ee.ImageCollection(imageCollection).select(dependent).count();
    // There are n-p degrees of freedom
    var dof = n.subtract(Independents.length());
    // Get the root mean square residuals in a pixel.
    var rmsr = linearTrend.select('residuals')
        .arrayProject([0])
        .arrayFlatten([
            ['rmsr']
        ]);
    // Residual sum of squares.
    var rss = rmsr.pow(2).multiply(n);
    // Estimated variance = RSS/(n-p).
    var sSquared = rss.divide(dof);
    // Sample variance of the dependent variable.
    var yVariance = imageCollection.select(dependent)
        .reduce(ee.Reducer.sampleVariance());
    // Adjusted, squared, multiple regression coefficient.
    var rSquareAdj = ee.Image(1).subtract(sSquared.divide(yVariance));
    var theRet = {}
    theRet.theResiduals = rmsr
    theRet.theRMSE = rmsr
    theRet.theR2 = rSquareAdj
    return (theRet)
}
function afn_fitHarmonicToCollection(harmonicIC, trackedBand, harmonicIndependents) {
    harmonicIC = ee.ImageCollection(harmonicIC)
    var dependent = ee.String(trackedBand);
    var harmonicIndependents = ee.List(harmonicIndependents);
    // Fit the model.
    var harmonicTrend = harmonicIC
        .select(harmonicIndependents.add(dependent))
        // The output of this reducer is a 4x1 array image.
        .reduce(ee.Reducer.linearRegression(harmonicIndependents.length(),
            1));
    //print("999999 the harmonic trend", harmonicTrend)
    // Turn the array image into a multi-band image of coefficients.
    var harmonicTrendCoefficients = harmonicTrend.select('coefficients')
        .arrayProject([0])
        .arrayFlatten([harmonicIndependents]);
    var myrmsr = harmonicTrend.select('residuals')
    // .arrayProject([0])
    // .arrayFlatten([['rmsr']]);
    // myrmsr = ee.Image(33)
    var RMSEandR2 = afn_getRMSEandR2(harmonicTrend, harmonicIC, harmonicIndependents, trackedBand)
    var theReturn = {}
    theReturn.harmonicTrendCoefficients = harmonicTrendCoefficients
    theReturn.theResiduals = RMSEandR2.theResiduals
    // theReturn.theRMSE = RMSEandR2.theRMSE
    // theReturn.theRMSE = myrmsr
    theReturn.theR2 = RMSEandR2.theR2
    theReturn.harmonicTrendCoefficients = harmonicTrendCoefficients // appears duplicated from earlier line, but leaving it in.

    theReturn.linearTrendObject = harmonicTrend

    // jc added the below on 8/21/25
    theReturn.numberImages = harmonicIC.size()

    return (theReturn)
}
exports.afn_fitHarmonicToCollection = afn_fitHarmonicToCollection;
function afn_applyHarmonicToCollection(harmonicIC, trackedBand, harmonicIndependents, harmonicTrendCoefficients) {
    harmonicIC = ee.ImageCollection(harmonicIC)
    var dependent = ee.String(trackedBand);
    var harmonicIndependents = ee.List(harmonicIndependents)
    // Compute fitted values.
    var fittedHarmonic = harmonicIC.map(function(image) {
        return image.addBands(
            image.select(harmonicIndependents)
            .multiply(harmonicTrendCoefficients)
            .reduce('sum')
            .rename('fitted'));
    });
    return (fittedHarmonic)
}

var afn_addTimeBandUsingPropertyNameToIC = function(anIC, var_args_timeBand) {
    // just adds a time band to an entire IC for a given property value. each image will have to have that property.
    // so it might be fracDOY (a number like 0.113 for a day 11% of the way through year;
    //  or floatDate, a number like 6.113
    var dateProperty = var_args_timeBand.dateProperty
    var timeBandName = var_args_timeBand.timeBandName
    var afn_addFractionalYearFromPropertyName = function(anImage) {
        // ideas about period can be seen here, for example
        // http://jwilson.coe.uga.edu/EMAT6680/Dunbar/Assignment1/sine_curves_KD
        // b is the period of the sine curve
        // The period of the sine curve is the length of one cycle of the curve.  The natural period of the sine
        // curve is 2π.  So, a coefficient of b=1 is equivalent to a period of 2π.  To get the period of the sine
        // curve for any coefficient b, just divide 2π by the coefficient b to get the new period of the curve.
        var numberForEachPixel = ee.Number(anImage.get(dateProperty));
        var theTimeBand = ee.Image(numberForEachPixel).rename(timeBandName).float()
        return anImage
            .addBands(theTimeBand)
    }
    var theICreturn = ee.ImageCollection(anIC).map(afn_addFractionalYearFromPropertyName) //.map(afn_addCosAndSin)
    return (theICreturn)
}
exports.afn_addTimeBandUsingPropertyNameToIC = afn_addTimeBandUsingPropertyNameToIC

var afn_addCosAndSinBandsToIC = function(anIC, var_args_harmonicBands) {
    // Input parameters
    // assumes there is a layer with the given name timeBandName.
    var timeBandName = var_args_harmonicBands.timeBandName
    var nonStandardPeriodBoolean = var_args_harmonicBands.nonStandardPeriodBoolean
    var nonStandardPeriodValue = var_args_harmonicBands.nonStandardPeriodValue
    var cosName = var_args_harmonicBands.cosName
    var sinName = var_args_harmonicBands.sinName
    // Distantly derived from harmonics code from Nick's Earth Engine Summit session.
    // Determine the period
    if (nonStandardPeriodBoolean) {
        // BULC-D uses a number that is dayRange / a half year. So e.g., 155/182
        var period = nonStandardPeriodValue
        // var period = dayRange.divide(aHalfYear); //<-- 155/182ish
        // used by BULC-D
    } else {
        // standard period is 2.
        var period = 2 // smaller number is more cycles per a given period
        // 0.5 means 4 cycles per year
        // 1 means 2 cycles per year
        // 2 means 1 cycle per year
        // 4 means 1/2 cycle per year
        // a non-round number like 0.9 is somewhere in between.
    }
    // Calculate angular frequency
    var angularFrequency = ee.Number(2).divide(period); //<-- 2*1.174 = 2.3564
    // Compute time in radians
    var twoPiFreq = angularFrequency
        .multiply(2 * Math.PI);
    // Inner function to calculate cosine and sine factors for an image of a given time and angular frequency.
    var afn_getCosAndSinForDateGivenAngularFreq = function(anImage) {
        // ideas about period can be seen here, for example
        // http://jwilson.coe.uga.edu/EMAT6680/Dunbar/Assignment1/sine_curves_KD
        // b is the period of the sine curve
        // The period of the sine curve is the length of one cycle of the curve.  The natural period of the sine
        // curve is 2π.  So, a coefficient of b=1 is equivalent to a period of 2π.  To get the period of the sine
        // curve for any coefficient b, just divide 2π by the coefficient b to get the new period of the curve.
        // Get the specified time band from the image
        var theTimeBand = ee.Image(anImage).select(timeBandName)
        // Calculate time in radians
        var timeRadians = theTimeBand
            .multiply(twoPiFreq)
        // Calculate cosine and sine factors and give them the specified names.
        var theCosFactor = timeRadians.cos().rename(cosName) //cos() * 2πωt --> cos(2πωt)
        var theSinFactor = timeRadians.sin().rename(sinName) //sin() * 2πωt --> sin(2πωt)
        // Return the image with additional bands representing cosine and sine factors
        return anImage
            .addBands(theCosFactor)
            .addBands(theSinFactor)
    }
    // Apply the inner function to each image in the image collection
    var theICreturn = ee.ImageCollection(anIC).map(afn_getCosAndSinForDateGivenAngularFreq) //.map(afn_addCosAndSin)
    // Return the modified image collection
    return (theICreturn)
}
exports.afn_addCosAndSinBandsToIC = afn_addCosAndSinBandsToIC
