function afn_createPriorsFromLandcoverMap(thisrunsInitializingLeveler, initialImage, numberOfClassesToTrack, ProbsLabel) {
    print("Creating starting probabilities from base land cover map...")
    initialImage = ee.Image(initialImage).unmask().multiply(1) // the .multiply(1) is a hack as mentioned in the forum
    // on April 9 and 10 2020.
    // with that done, some pixels might be nodata on the initialImage.  This changes
    // them to be 1.
    //    initialImage = ee.Image(1).where(initialImage.gt(1),initialImage);


    // create a list of numbers from 1 to a number like '4'.
    // cast it to an array
    var theArrayList = ee.Array(ee.List.sequence(1, ee.Number(
        numberOfClassesToTrack)))
    //print(theArrayList, "217.theArrayList")

    // Now, create an array image from that array of values.
    // the intention is a 1D array of the numbers 1,2,3,4
    // (but where the number '4' is variable and passed in)

    var theStackAsArray = ee.Image(theArrayList).rename(ProbsLabel)
    // print("218.theStackAsArray", theStackAsArray)
    // Map.addLayer(theStackAsArray, {}, "218") // works fine.

    // a bit of leftover testing from April 10 2020 errors.
    // Now, we seek the following:
    // for each pixel, given that the value was i, set the ith
    // slot to 1 and the other slots to 0.

    // Try 1: hardcoded. good.
    // var theLCMapExploded = theStackAsArray.eq(2) // hardcoded value works

    // Try 2: set a constant image. Good.
    //var initialImage = ee.Image.constant(3) // constant image works

    // Try 3: random image set to betw 1 and 3, truncated and set to integer. Good.
    //var initialImage = ee.Image.random(0, "uniform").multiply(3).add(0.5).round().int()


    // show the initial image.
    // Map.addLayer(initialImage, {}, "216.Initial")

    var theLCMapExploded = theStackAsArray.eq(initialImage) // constant image works

    // Map.addLayer(theLCMapExploded, {}, "219.theLCMapExploded")
    // print("335.exploded map with 1s", theLCMapExploded)
    // print("Finished creating starting probabilities.")

    // now let's level that exploded map
    var theLCMapExplodedAndLeveled = afn_LevelInitialLCExploded(theLCMapExploded, thisrunsInitializingLeveler, numberOfClassesToTrack)
    // Map.addLayer(theLCMapExplodedAndLeveled, {}, "336b.starting probs")

    return (theLCMapExplodedAndLeveled);
}

function afn_LevelInitialLCExploded(theLCMapExploded, thisrunsInitializingLeveler, numberOfClassesToTrack) {
    // implement theLCMapExploded * thisrunsInitializingLeveler + (1-thisrunsInitializingLeveler)/numberOfClassesToTrack
    var levelingMinimumNumeratorPart1 = thisrunsInitializingLeveler.multiply(-1)
    var levelingMinimumNumerator = levelingMinimumNumeratorPart1.add(1)
    var levelingMinimum = levelingMinimumNumerator.divide(numberOfClassesToTrack)

    var theRet = theLCMapExploded.multiply(thisrunsInitializingLeveler).add(levelingMinimum);
    //print ("336.starting probs", theRet)
    return (theRet)
}

function afn_create1DTransitionArrayImageFromHardCodedTruth(oneEvent, anyDampenedTruthTable, listOfEventClassesFrom0, condStackLabel) {
    // accesses an already-existing 2d hardcoded truth table.
    // index the rows of that table based on the oneEvent value
    // Returns a 1D array image

    var TTasList = ee.List(anyDampenedTruthTable) //
    // Now, pull out the right set of conditional values based on the oneEvent value.
    var TruthTableRowToArrayImage = ee.Image(oneEvent).remap(listOfEventClassesFrom0, TTasList).rename([condStackLabel])

    return (TruthTableRowToArrayImage)
}



function afn_create1DTransitionArrayImageFromDynamicTruth(oneEvent, anyDampenedTruthTable, listOfEventClassesFrom0, condStackLabel) {
    // accesses a passed-in already-dampened truth table.
    // index the rows of that table based on the oneEvent value
    // Returns a 1D array image
    listOfEventClassesFrom0 = ee.List(listOfEventClassesFrom0)

    function afn_sliced2DtoRemappable(oneSlicedTruthTable2D) {
        // the following works: treat the table by casting to array, then to list, then map over
        // the list and return each row as an array.  Seems crazy but works!
        var the2dAsList = ee.Array(oneSlicedTruthTable2D).toList()
        var the2DTransition = the2dAsList.map(function(l) {
            return ee.Array(l)
        })
        return (the2DTransition)
    }
    var anyDampenedTruthTableAsListOfArrays = afn_sliced2DtoRemappable(anyDampenedTruthTable)

    // Now, pull out the right set of conditional values based on the oneEvent value.
    var TruthTableRowToArrayImage = ee.Image(oneEvent).remap(listOfEventClassesFrom0, anyDampenedTruthTableAsListOfArrays).rename([condStackLabel])

    return (TruthTableRowToArrayImage)
}

function afn_extractNamedSlot(theMBStack, layerName) {
    return ee.Image(theMBStack).select(layerName);

function afn_overwriteNamedSlot(theMBStack, layerValues, layerName) {
    return ee.Image(theMBStack).addBands(layerValues, [layerName], true); // it overwrites bc it is parameterized as 'true'
}

function afn_buildStartProbs(bbiidd) {
    //  print("66432.building start probs", bbiidd)
    // unpack the parameter dictionary
    var initializationApproach = bbiidd.initializationApproach
    // print(initializationApproach)
    var initializingLeveler = bbiidd.initializingLeveler
    // print(initializingLeveler)
    var baseLandCoverImage = bbiidd.baseLandCoverImage
    // print(baseLandCoverImage)

    // print(numberOfClassesToTrack)

    // print(defaultStudyArea)

    // print(classNameList)

    // var SPECIALinitializingLeveler = bbiidd.SPECIALinitializingLeveler

    // var SPECIALnumberOfClassesToTrack = bbiidd.SPECIALnumberOfClassesToTrack

    // unpack the parameter dictionary


    // first, we'll just do the base map set up of those probabilities.
    if (initializationApproach === "F") { // F meaning a first run image.
        // initializationApproachParameter1 : holds the base image used to initialize the land cover map.
        // initializationApproachParameter2: holds the comparison layer to start (if Overlaying). can be different than the initialization.
        // see below these ifs for the comparison layer code.

        var currentProbsArrayImg = ee.Image(afn_createPriorsFromLandcoverMap(initializingLeveler, baseLandCoverImage, numberOfClassesToTrack, ProbsLabel))

        // Map.addLayer(currentProbs,{},"q13435b.current probs")
        // print("set.")
    }
    if (initializationApproach === "A") { // A meaning Read from Asset.
        // initializationApproachParameter1: holds the string name of the asset with the ending probabilities from an earlier run
        // initializationApproachParameter2: holds the string name of the asset with the ending comparison layer from an earlier run

        // read an asset. Asset name comes in initializationApproachParameter1 as a string..
        var currentProbsMultiBand = afn_ImportMultiBandAssetFromEarlierIteration(initializationApproachParameter1, classNameList);

    }
    if (initializationApproach === "S") { // S meaning 'sequential' from an iterate inside this same run that hasn't been saved to an asset.

        // here AccumulatingAnswer will be sent in import.  the probs are stored as an array there so no need to cast them after.

        // initializationApproachParameter1 : holds the computed object from within this run. used for chaining iterates. Contains iterim latest

        // initializationApproachParameter2: null. Not used.

        print("2164.getting currentProbs in Serial Input")

        print("2165.currentProbs in Serial Input", currentProbs)
        Map.addLayer(currentProbs, {}, "2349s.Beginning Probability MultiBand", true, 0)

    }
    return (currentProbs)

}

function afn_buildComparisonLayer(cciidd) {

    // unpack the parameter dictionary
    var transitionCreationMethod = cciidd.transitionCreationMethod
    var comparisonLayerLabel = cciidd.comparisonLayerLabel
    var initializationApproach = cciidd.initializationApproach
    var initializationApproachParameter1 = cciidd.initializationApproachParameter1
    var firstComparisonImage = cciidd.firstComparisonImage
    var defaultStudyArea = cciidd.defaultStudyArea
    // done unpacking the parameter dictionary




    // Now, develop the comparison layer if it is needed.
    if (transitionCreationMethod === "C") { //  meaning the transitions use a custom-passed matrix. We don't need the comparison layer. we'll set it to -1.
        var theComparisonLayer = ee.Image(-1).rename(comparisonLayerLabel)
    }

    if (transitionCreationMethod === "D") { //  meaning the transitions use the identify matrix. We don't need the comparison layer. we'll set it to -1.
        var theComparisonLayer = ee.Image(-1).rename(comparisonLayerLabel)
    }

    // If we are doing the overlay approach
    if (transitionCreationMethod === "V") { // V meaning we are doing overlays. We need to know more. Is this a first run or a continuing, or an asset read?
        if (initializationApproach === "F") { //  meaning a first run image.
            // in this case, it's in another parameter
            var theComparisonLayer = ee.Image(firstComparisonImage).rename([comparisonLayerLabel]) //.clip(defaultStudyArea)
        }
        if (initializationApproach === "A") { //  meaning Read from Asset.
            // initializationApproachParameter1: holds the string name of the asset with the ending probabilities from an earlier run
            // initializationApproachParameter2: holds the string name of the asset with the ending comparison layer from an earlier run

            // Again, read an asset for the comparison layer. comes in initializationApproachParameter2 as a string..
            var theComparisonLayer = afn_ImportMultiBandAssetFromEarlierIteration(initializationApproachParameter2, [comparisonLayerLabel])
            print("importing Comparison Layer from previous iteration with name: " + initializationApproachParameter2)
        }
        if (initializationApproach === "S") { // S meaning 'sequential' from an iterate inside this same run that hasn't been saved to an asset.
            // serial input.  will be sent the old prior probs in inputOptional as a multi-band.
            // here AccumulatingAnswer will be sent in import.  the probs are stored as an array there so no need to cast them after.

            // initializationApproachParameter1 : holds the computed object from within this run. used for chaining iterates. Contains iterim latest
            // probabilities and latest comparison layer.
            var theComparisonLayer = afn_extractNamedSlot(notImplementedYet, comparisonLayerLabel)
        }
    }
    return (theComparisonLayer)
}

    }

    // add 1 to the local image counter, in every pixel with a valid value in this iteration.
    thePerPixelImageCounter = thePerPixelImageCounter.where(oneEventValidValues, thePerPixelImageCounter.add(1)).rename([PerPixelImageCounterLabel]);
    accumulatingAnswer = afn_overwriteNamedSlot(accumulatingAnswer, thePerPixelImageCounter, PerPixelImageCounterLabel)

    // getting the Event this iteration.  Note that it is passed in on the stack of images used for the iterate. Here it is called oneEvent.
    // below we write it to the structure at the bottom, for testing.
    if (RecordEventsAtEachTimeStep) {
        accumulatingAnswer = accumulatingAnswer.addBands(oneEvent.mask(oneEventValidValues).rename([EventLabel]), [EventLabel], false)
    }

    if (transitionCreationMethod === "C") {
        // in this case, truthTableStack arrives as a single custom-passed matrix with as many rows as the max classes in events.
        // this can be dampened using the transition leveler, if desired
        //print("3316. truth table stack", truthTableStack)
        var transitionArrayImageRescaledAndDampened = afn_create1DTransitionArrayImageFromDynamicTruth(oneEvent, truthTableStack,
            listOfEventClassesFrom0, condStackLabel)
    }

    if (transitionCreationMethod === "D") {
        // in this case, truthTableStack arrives as a single dampened identity matrix with a dummy row of 1s on top.
        //var oneDampenedTruthTable = truthTableStack
        var transitionArrayImageRescaledAndDampened = afn_create1DTransitionArrayImageFromHardCodedTruth(oneEvent, truthTableStack,
            listOfEventClassesFrom0, condStackLabel)
        accumulatingAnswer = accumulatingAnswer.set('LatestTransition', truthTableStack)
    }

    if (transitionCreationMethod === "V") {
        // pull out one from the stack of truth tables
        // here, the truth table stack is a stack of truth tables. each is a dampened matrix that had a row of 1s put on top.
        // this was done before ever entering the iterate. So here we just pull the rightly indexed one out of the stack.
        var whichDynamicTableToGet = ee.Number(accumulatingAnswer.get(globalCounterPropertyName))
        //print(whichDynamicTableToGet)

        var oneDampenedTruthTable = ee.List(truthTableStack).get(whichDynamicTableToGet)
        var transitionArrayImageRescaledAndDampened = afn_create1DTransitionArrayImageFromDynamicTruth(oneEvent, oneDampenedTruthTable,
            listOfEventClassesFrom0, condStackLabel)
        accumulatingAnswer = accumulatingAnswer.set('LatestTransition', oneDampenedTruthTable)
        //accumulatingAnswer = accumulatingAnswer.set(ee.List(['LatestTransition']).cat(ee.Number.parse(whichDynamicTableToGet)), oneDampenedTruthTable)
    }
    // now, however it was made, pull from the right row of that table to make the transition array image.
    // accumulatingAnswer = accumulatingAnswer.set(ee.String.encodeJSON(ee.Number.parse(whichDynamicTableToGet)), oneDampenedTruthTable)
    // accumulatingAnswer = accumulatingAnswer.set(ee.String.encodeJSON(whichDynamicTableToGet), oneDampenedTruthTable)


    // other accessing the table here
    // At this point, the transition array image is created.  Write it if requested
    if (RecordConditionalsAtEachTimeStep) {
        accumulatingAnswer = accumulatingAnswer.addBands(
            transitionArrayImageRescaledAndDampened) //, conditionalsNameList, false) // write to bottom of the stack as an array band. Intended for testing.
    }

    // Bayes application here
    // Handy Bayes one-liner
    var posteriorProbsValidPixels1D = currentProbs.mask(oneEventValidValues).multiply(transitionArrayImageRescaledAndDampened).divide(currentProbs
        .arrayDotProduct(transitionArrayImageRescaledAndDampened))
    // for testing: accumulatingAnswer = accumulatingAnswer.addBands(posteriorProbsValidPixels1D, [ProbsLabel]) // write posterior probs into the array image at slot 0

    // rebalance probabilities
    // dampen the resulting probs
    posteriorProbsValidPixels1D = ee.Image(afn_dayIRebalancingV3(posteriorProbsValidPixels1D, posteriorLeveler, posteriorMinimum))


    // burn the pixel with updated values onto the prior probs. Result should have value at every pixel.
    var posteriorProbsAllPixels1D = ee.Image(currentProbs.where(oneEventValidValues, posteriorProbsValidPixels1D)).rename([ProbsLabel])

    accumulatingAnswer = accumulatingAnswer.addBands(posteriorProbsAllPixels1D, [ProbsLabel], true) // write posterior probs into the array image at slot 0
    if (RecordProbabilitiesAtEachTimeStep) {
        accumulatingAnswer = accumulatingAnswer.addBands(posteriorProbsAllPixels1D.arrayFlatten([classNameList])) // write posterior probs into the array image at slot 0
    }

    if (RecordBULCLayersAtEachTimeStep) {
        var BULCposterior = posteriorProbsAllPixels1D.arrayArgmax().arrayFlatten([
            [BULCLayerLabel]
        ]).add(1) // add 1 bc argmax returns on 0 scale.
        accumulatingAnswer = accumulatingAnswer.addBands(BULCposterior, [BULCLayerLabel], false) // write posterior probs into the array image at slot 0
    }
    if (RecordConfidenceAtEachTimeStep) {
        var BULCConfidence1DArray = posteriorProbsAllPixels1D.arrayReduce(ee.Reducer.max(), [0]).select(0)
        var BULCConfidence0D = BULCConfidence1DArray
            .reduce(ee.Reducer.max()).arrayFlatten([
                ['BULCconfidenceLabel']
            ])
        accumulatingAnswer = accumulatingAnswer.addBands(BULCConfidence0D) //, [BULCconfidenceLabel], false) // write posterior probs into the array image at slot 0
    }

    //increase the counter that indicates how many images we have seen so far.
    accumulatingAnswer = ee.Image(afn_incrementGlobalImageCounter(accumulatingAnswer, globalCounterPropertyName))

    return accumulatingAnswer
}
