function afn_gatheringArgsParking(gatheringCollections_args) {
/* * This function creates a deep copy of the arguments passed in the `gatheringCollections_args` object.
 * It's necessary because JavaScript's assignment operator (x = y) performs a shallow copy, meaning
 * changes to nested objects in the copy will also affect the original. To avoid this, we
 * serialize and then re-parse each property, creating a truly independent copy.
 *
 * @param {object} gatheringCollections_args - The object containing the arguments to be copied.
 * @returns {object} - A deep copy of the input arguments.
 */

    // Create an empty object to store the copied arguments.
    var parkedArgs = {};
    // Copy each property by serializing and then re-parsing it. This creates a deep copy.
    parkedArgs.bandNameToFit = JSON.parse(JSON.stringify(gatheringCollections_args.bandNameToFit));
    parkedArgs.bandName_reduction = JSON.parse(JSON.stringify(gatheringCollections_args.bandName_reduction));
    parkedArgs.binCuts = JSON.parse(JSON.stringify(gatheringCollections_args.binCuts));
    parkedArgs.centeringZoom = JSON.parse(JSON.stringify(gatheringCollections_args.centeringZoom));
    parkedArgs.expectationCollectionParameters = JSON.parse(JSON.stringify(gatheringCollections_args.expectationCollectionParameters));
    parkedArgs.modalityDictionary = JSON.parse(JSON.stringify(gatheringCollections_args.modalityDictionary));
    parkedArgs.plottingMeans = JSON.parse(JSON.stringify(gatheringCollections_args.plottingMeans));
    parkedArgs.targetCollectionParameters = JSON.parse(JSON.stringify(gatheringCollections_args.targetCollectionParameters));
    parkedArgs.verbose = JSON.parse(JSON.stringify(gatheringCollections_args.verbose));
    parkedArgs.whichReduction = JSON.parse(JSON.stringify(gatheringCollections_args.whichReduction));
    // The `defaultStudyArea` property is a special server object, so we can't stringify it.
    // Instead, we directly assign it to the copied object.
    parkedArgs.expectationCollectionParameters.defaultStudyArea = gatheringCollections_args.expectationCollectionParameters.defaultStudyArea;
    parkedArgs.targetCollectionParameters.defaultStudyArea = gatheringCollections_args.targetCollectionParameters.defaultStudyArea;
    // Return the copied object.
    return (parkedArgs);
}

            expectationResiduals: expectationResiduals, // observed - fitted for expectation
            expectationPeriodSummaryValue: expectationPeriodSummaryValue, // 7. Image. Summary statistic for the expectation period.
            expectationPeriodSD: expectationPeriodSD, // 8. Image. Standard deviation of the expectation period.
            // expectationPeriodDeviationsIC: expectationPeriodDeviationsIC, // not checked closely. better to use expectationresiduals instead. ImageCollection. Image collection of deviations from the fitted model for the expectation period.
            //                                                               // Three bands: Observed, Expected, Difference Between Them.
            expectationPeriodMAD: expectationPeriodMAD, // 10. Image. Median Absolute Deviation of the expectation period.
            targetCollection: targetCollection, // 11. ImageCollection. The gathered and processed target image collection.
            targetCollectionFit: targetCollectionFit, // 12. ImageCollection. The target collection with the fitted harmonic model applied.
            targetResiduals: targetResiduals, // observed - fitted for target
            targetCollectionLackOfFit: targetCollectionLackOfFit, // 13. Multi-band Image. The difference between target and fitted values.
            targetLOFAsZScore: targetLOFAsZScore, // 14.  Multi-band Image. The normalized lack of fit of the target collection (Z-scores) calculated using residual standard deviation.
            targetPeriodSummaryValue: targetPeriodSummaryValue // 15. Image. Summary statistic for the target period.
            // theExpectationRMSE : theExpectationRMSE,
        } // end of afn_createExpAndTargetStream
    }
    var hasOptical = afn_checkForOptical(var_args.expectationCollectionParameters)
    var hasSAR = afn_checkForSAR(var_args.expectationCollectionParameters)
    // print(hasSAR, "has SAR1")
    print("**************************************")
    print(hasOptical, "has Optical")
    print(hasSAR, "has SAR2")
    var blendingStreams = hasOptical && hasSAR;
    print("blending streams?", blendingStreams)
    // var blendingStreams = var_args.blendingStreams // to set within the interface. comment this in when ready.
    if (!blendingStreams) {
        print("Not blending streams. Happens when only one of Optical or SAR is checked.")
        var organizedBULCDStream = afn_createExpAndTargetStream(var_args)
        print("organizedBULCDStream is", organizedBULCDStream)
    }
    if (blendingStreams) {
        print("Yes, blending streams. This happens when both Optical and SAR are checked.")
        // Isolate the SAR part and analyze its data stream.
        // Park the arguments because of how JS doesn't make a new copy of dictionaries. This is analogous to a deep copy.
        var parked_args = afn_gatheringArgsParking(var_args)
        print("parked args", parked_args)
        // now we can work with the var_args dictionary, knowing we can reinstate it at the end.
        // var_args.aaa = 1 // for debugging

        // let's turn off the Optical Parameters while we collection the SAR values.
        // note we might have S1 on, or might have AL on, or NISAR on. Our goal is to turn off the opticals and
        // leave the SARS as-is.
        var_args.expectationCollectionParameters.datasetSelection.L5 = false;
        var_args.expectationCollectionParameters.datasetSelection.L7 = false;
        var_args.expectationCollectionParameters.datasetSelection.L8 = false;
        var_args.expectationCollectionParameters.datasetSelection.L9 = false;
        var_args.expectationCollectionParameters.datasetSelection.MO = false;
        var_args.expectationCollectionParameters.datasetSelection.S2 = false;
        var_args.targetCollectionParameters.datasetSelection.L5 = false;
        var_args.targetCollectionParameters.datasetSelection.L7 = false;
        var_args.targetCollectionParameters.datasetSelection.L8 = false;
        var_args.targetCollectionParameters.datasetSelection.L9 = false;
        var_args.targetCollectionParameters.datasetSelection.MO = false;
        var_args.targetCollectionParameters.datasetSelection.S2 = false;
        print("modified collection Arguments: SAR", var_args)
        var SARStream = afn_createExpAndTargetStream(var_args)
        print("SAR Stream", SARStream)
        // Now, do optical. start by restoring the original var_args.
        print("now doing optical...")
        var_args.expectationCollectionParameters = parked_args.expectationCollectionParameters
        var_args.targetCollectionParameters = parked_args.targetCollectionParameters
        print("restored var_args version 2", var_args)

        // need to flip the radar ones so that the opticals are gathered correctly.
        var_args.expectationCollectionParameters.datasetSelection.AL = false;
        var_args.targetCollectionParameters.datasetSelection.AL = false;
        var_args.expectationCollectionParameters.datasetSelection.NI = false;
        var_args.targetCollectionParameters.datasetSelection.NI = false;
        var_args.expectationCollectionParameters.datasetSelection.S1 = false;
        var_args.targetCollectionParameters.datasetSelection.S1 = false;
        print("modified collection Arguments: Optical", var_args)
        var opticalStream = afn_createExpAndTargetStream(var_args)
        print("Optical Stream", opticalStream)
        // now, restore var_args in case that is needed downstream.
        var_args.expectationCollectionParameters = parked_args.expectationCollectionParameters
        var_args.targetCollectionParameters = parked_args.targetCollectionParameters
        print("restored var_args version 3", var_args)
        print("Now, need to combine the streams!")
        print("opticalStream.targetLOFAsZScore", opticalStream.targetLOFAsZScore)
        print("SARStream.targetLOFAsZScore", SARStream.targetLOFAsZScore)
        var organizedBULCDStream = afn_blendOpticalAndSARStreams(opticalStream, SARStream)
    }
    print("organizedBULCDStream", organizedBULCDStream)
    return {
        // commentary from Gemini. truly amazing.
        expectationCollection: organizedBULCDStream.expectationCollection, // 1. ImageCollection. Gathered and processed expectation collection.
        expectationYearRegressionObject: organizedBULCDStream.expectationYearRegressionObject, // 2. Object. Object containing details about the fitted harmonic regression model to expectation data.
        aCoefficientSetExpectationYear: organizedBULCDStream.aCoefficientSetExpectationYear, // 3. Image. Coefficients of the fitted harmonic model (expectation).
        // theExpectationResiduals: theExpectationResiduals, // 4. ImageCollection. Image collection of residuals from the fitted harmonic model (expectation).
        theExpectationR2: organizedBULCDStream.theExpectationR2, // 5. Image. R-squared value of the fitted harmonic model (expectation).
        expectCollectionFit: organizedBULCDStream.expectCollectionFit, // 6. ImageCollection. The expectation collection with the fitted harmonic model applied.
        expectationResiduals: organizedBULCDStream.expectationResiduals, // observed - fitted for expectation
        expectationPeriodSummaryValue: organizedBULCDStream.expectationPeriodSummaryValue, // 7. Image. Summary statistic for the expectation period.
        expectationPeriodSD: organizedBULCDStream.expectationPeriodSD, // 8. Image. Standard deviation of the expectation period.
        // expectationPeriodDeviationsIC: expectationPeriodDeviationsIC, // not checked closely. better to use expectationresiduals instead. ImageCollection. Image collection of deviations from the fitted model for the expectation period.
        //                                                               // Three bands: Observed, Expected, Difference Between Them.
        expectationPeriodMAD: organizedBULCDStream.expectationPeriodMAD, // 10. Image. Median Absolute Deviation of the expectation period.
        targetCollection: organizedBULCDStream.targetCollection, // 11. ImageCollection. The gathered and processed target image collection.
        targetCollectionFit: organizedBULCDStream.targetCollectionFit, // 12. ImageCollection. The target collection with the fitted harmonic model applied.
        targetResiduals: organizedBULCDStream.targetResiduals, // observed - fitted for target
        targetCollectionLackOfFit: organizedBULCDStream.targetCollectionLackOfFit, // 13. Multi-band Image. The difference between target and fitted values.
        targetLOFAsZScore: organizedBULCDStream.targetLOFAsZScore, // 14.  Multi-band Image. The normalized lack of fit of the target collection (Z-scores) calculated using residual standard deviation.
        targetPeriodSummaryValue: organizedBULCDStream.targetPeriodSummaryValue // 15. Image. Summary statistic for the target period.
        // theExpectationRMSE : theExpectationRMSE,
    }
}
