 /**
  * 1. Cloud Masking and Scaling Functions:
  *
  *   - The code defins a series of functions for cloud masking and applying scale factors to
  *     different satellite imagery datasets:
  *       - `maskLowQA`:  Masks pixels in Sentinel-2 imagery based on Cloud Score Plus QA values
  *         (for images from 2021 onwards).
  *       - `maskSrCloudsL8andL9`: Masks clouds and saturation in Landsat 8 and 9 imagery using QA bands.
  *       - `afn_cloudMaskIC_L5andL7`: Masks clouds and shadows in Landsat 5 and 7 imagery using QA bands.
  *       - `afn_cloudMaskIC_MO`: Placeholder for potential cloud masking in MODIS imagery
  *         (currently does nothing).
  *       - `afn_cloudMaskIC_olderS2`: Masks clouds, cirrus, and water in Sentinel-2 imagery
  *         (for images before 2019) using QA bands.
  *       - `applyScaleFactorsL789`: Applies scale factors to Landsat 7, 8 and 9 surface reflectance
  *         and thermal bands.
  *       - `applyScaleFactorsS2`: Applies scale factors to Sentinel-2 bands.
  *
  * 2. `afn_gatherCollectionsAndReduce` Function:
  *
  *   - This is the core function that gathers image collections from various sensors, filters them,
  *     applies cloud masking, performs data reduction, and merges the results.
  *   - Arguments (`gatherCollections_args`):
  *      - `datasetSelection`: A dictionary specifying which sensors to include (L5, L7, L8, L9, MO, S1, S2, DW).
  *      - `defaultStudyArea`:  The geographic region of interest.
  *      - `dayStepSize`: The time interval (in days) between image acquisitions.
  *      - Individual sensor dictionaries (e.g., `L5dictionary`, `S2dictionary`) containing
  *        sensor-specific parameters like years, date ranges, and cloud cover thresholds.
  *      - `bandName_reduction`: The name of the band to use for reduction.
  *      - `whichReduction`: The type of reduction to apply (e.g., "NBR", "NDVI", "count", "mode").
  *      - `verbose`: A flag to control the amount of output printed.
  *
  *   - Key Steps:
  *       - Sensor Selection:  The code determines which sensors are activated in `datasetSelection`
  *         and initializes corresponding variables.
  *       - Date Range Determination: It finds the earliest start date, latest end date, and the
  *         full set of years to process based on the selected sensors' parameters.
  *       - Nested Functions for Data Reduction:  It defines nested functions for calculating NBR,
  *         NDVI, SWIR1, SWIR2, and other values for different sensors (L5, L7, L8, L9, MO, S1, S2).
  *       - Iterating Through Years and Days: The code iterates through the selected years and
  *         creates an image collection with one image per `dayStepSize` interval.
  *       - Filtering and Processing: For each day, it filters the image collections based on
  *         date, cloud cover, and applies the appropriate cloud masking and scaling functions.
  *       - Data Reduction: It applies the specified reduction method (`whichReduction`) to the
  *         filtered image collections for each day (e.g., calculating the median NBR, counting images,
  *         or determining the mode of a classification).
  *       - Merging Results: It merges the results from all selected sensors into a single image
  *         collection.
  *
  * 3. Testing Sections:
  *
  *   - The code includes two testing sections (`testing1` and `testing2`) that demonstrate how to
  *     use `afn_gatherCollectionsAndReduce` with different sensor configurations and reduction methods.
  *
  */

 // Function to mask pixels with low CS+ QA scores.
 function maskLowQA(image) {
     var qaBand = 'cs';
     var clearThreshold = 0.60;
     var mask = image.select(qaBand).gte(clearThreshold);
     return image.updateMask(mask);
 }
 // Define the cloud mask function. From checkpoint F43a in the book.
 function maskSrCloudsL8andL9(image) {
     // Bit 0 - Fill
     // Bit 1 - Dilated Cloud
     // Bit 2 - Cirrus
     // Bit 3 - Cloud
     // Bit 4 - Cloud Shadow
     var qaMask = image.select('QA_PIXEL').bitwiseAnd(parseInt('11111',
         2)).eq(0);
     var saturationMask = image.select('QA_RADSAT').eq(0);
     return image.updateMask(qaMask)
         .updateMask(saturationMask);
 }
 function afn_cloudMaskIC_L5andL7(inputImage) {
     var QA_Band_Name_Input = 'QA_PIXEL';
     //LANDSAT cloud masking function, embedded so that the QA can be set
     // within this scope, and not need to be a global variable.
     function afn_CloudMasked_NestedScope(inputImage) {
         inputImage = ee.Image(inputImage)
         var cloudShadowBitMask = 1 << 3
         var cloudsBitMask = 1 << 4
         // Get the pixel QA band.
         var qa = inputImage.select(QA_Band_Name_Input)
         // Both flags should be set to zero, indicating clear conditions.
         var mask = qa.bitwiseAnd(cloudShadowBitMask).eq(0)
             .and(qa.bitwiseAnd(cloudsBitMask).eq(0))
         // Return the masked image, scaled to TOA reflectance, without the QA bands.
         return inputImage.updateMask(mask)
             .copyProperties(inputImage, ["system:time_start"])
     }
     var imageToReturn = afn_CloudMasked_NestedScope(inputImage)
     return (imageToReturn)
 }

     return (imageToReturn)
 }
 function afn_cloudMaskIC_olderS2(inputImage) {
     function afn_CloudMasked_NestedScope(inputImage, params) {
         inputImage = ee.Image(inputImage)
         var Cloud_BQA = inputImage.select(params.QA_Band_Name_Input);
         var Cloud_Mask =
             Cloud_BQA.bitwiseAnd(params.cloudBit).eq(
                 params.clouds_bit_Thresh).and( // no clouds
                 Cloud_BQA.bitwiseAnd(params.cirrusBit).eq(
                     params.cirrus_bit_Thresh)).and( // no cirrus
                 Cloud_BQA.bitwiseAnd(params.waterBit).eq(
                     params.water_bit_Thresh)); // no water
         return inputImage.updateMask(Cloud_Mask);
     }
     var params = {
         QA_Band_Name_Input: 'QA60',
         cloudBit: Math.pow(2, 10),
         cirrusBit: Math.pow(2, 11),
         waterBit: Math.pow(2, 11),
         clouds_bit_Thresh: 0,
         cirrus_bit_Thresh: 0,
         water_bit_Thresh: 0
     }
     var imageToReturn = afn_CloudMasked_NestedScope(inputImage, params)
     return (imageToReturn)
 }

 function applyScaleFactorsL789(image) {
     var opticalBands = image.select('SR_B.').multiply(0.0000275).add(-0.2);
     var thermalBands = image.select('ST_B.*').multiply(0.00341802).add(149.0);
     return image.addBands(opticalBands, null, true)
         .addBands(thermalBands, null, true);
 }
 function applyScaleFactorsS2(image) {
     return image.addBands(image.select('B.').multiply(0.0001));
 }
